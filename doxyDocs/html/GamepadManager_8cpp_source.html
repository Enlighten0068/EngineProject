104void GamepadManager::AddGamepad(SDL_JoystickID deviceID){
105 if(!SDL_IsGamepad(deviceID)) return;
106
107 SDL_Gamepad* gamepad = SDL_OpenGamepad(deviceID);
108 if(!gamepad){
109 Log::Error(std::format("Failed to open gamepad {}: {}", deviceID, SDL_GetError()));
110 return;
111 }
112
113 //Find first available player slot
114 int playerIndex = 0;
115 while(playerIndex < MAX_PLAYERS && m_Gamepads.find(playerIndex) != m_Gamepads.end()){
116 playerIndex++;
117 }
118
119 if(playerIndex >= MAX_PLAYERS){
120 Log::Warning("Maximum number of gamepads reached.");
121 SDL_CloseGamepad(gamepad);
122 return;
123 }
124
125 //Store gamepad and initialize button states
126 m_Gamepads[playerIndex] = gamepad;
127 m_ButtonState[playerIndex][playerIndex] = 0;
128 m_ButtonPrevious[playerIndex][playerIndex] = 0;
129
130 const char* name = SDL_GetGamepadName(gamepad);
131 Log::Info(std::format("Gamepad {} connected: {} (Player {})", deviceID, name ? name : "Unknown", playerIndex));
132}

158void GamepadManager::UpdateButtonState(int playerIndex){
159 auto it = m_Gamepads.find(playerIndex);
160 if(it == m_Gamepads.end()) return;
161
162 SDL_Gamepad* gamepad = it->second;
163 Uint64 state = 0;
164
165 //Build bitmask of all buttons
166 for(int i = 0; i < SDL_GAMEPAD_BUTTON_COUNT; ++i){
167 if(SDL_GetGamepadButton(gamepad, static_cast<SDL_GamepadButton>(i))){
168 state |= (1ULL << i);
169 }
170 }
171 m_ButtonState[playerIndex][playerIndex] = state;
172}

178bool GamepadManager::IsButtonPressed(int playerIndex, SDL_GamepadButton button) const{
179 auto itState = m_ButtonState[playerIndex].find(playerIndex);
180 auto itPrev = m_ButtonPrevious[playerIndex].find(playerIndex);
181 if(itState == m_ButtonState[playerIndex].end() || itPrev == m_ButtonPrevious[playerIndex].end()) return false;

183 Uint64 mask = 1ULL << static_cast<int>(button);

196void BaseLevelScene::ResolveCollisions(){
197 auto& registry = m_ECSScene->GetRegistry();
198 auto& playerTransform = registry.get<Components::Transform>(m_PlayerEntity);
199 Vector3D playerHalfSize = playerTransform.Scale * 0.5f;
200
201 bool grounded = false;
202 const float GROUND_MARGIN = 0.1f;
203
204 float playerLeft = playerTransform.Position.x - playerHalfSize.x;
205 float playerRight = playerTransform.Position.x + playerHalfSize.x;
206 float playerBottom = playerTransform.Position.y - playerHalfSize.y;
207 float playerTop = playerTransform.Position.y + playerHalfSize.y;
208
209 for(entt::entity platform : m_PlatformEntities){
210 auto& pt = registry.get<Components::Transform>(platform);
211 Vector3D pHalf = pt.Scale * 0.5f;
212
213 float pLeft = pt.Position.x - pHalf.x;
214 float pRight = pt.Position.x + pHalf.x;
215 float pTop = pt.Position.y + pHalf.y;
216 float pBottom = pt.Position.y - pHalf.y;
217
218 bool overlapX = (playerRight > pLeft && playerLeft < pRight);
219 bool overlapY = (playerTop > pBottom && playerBottom < pTop);
220
221 if(overlapX){
222 float distanceToTop = playerBottom - pTop;
223 if(distanceToTop <= GROUND_MARGIN && distanceToTop >= -GROUND_MARGIN){
224 if(m_PlayerController->GetVelocity().y <= 0.0f){
225 grounded = true;
226 playerTransform.Position.y = pTop + playerHalfSize.y;
227 m_PlayerController->GetVelocity().y = 0.0f;
228 break;
229 }
230 }
231 }
232
233 if(!grounded && overlapX && overlapY){
234 Vector3D delta = playerTransform.Position - pt.Position;
235 Vector3D overlap = playerHalfSize + pHalf -
236 Vector3D(std::abs(delta.x), std::abs(delta.y), 0.0f);
237
238 if(overlap.x > 0.0f && overlap.y > 0.0f){
239 if(overlap.x < overlap.y){
240 //Horizontal resolution (player and platform overlap more vertically)
241 if(delta.x > 0.0f) playerTransform.Position.x += overlap.x;
242 else playerTransform.Position.x -= overlap.x;
243 } else{
244 //Vertical resolution (player and platform overlap more horizontally)
245 if(delta.y > 0.0f){
246 //Player center is above platform center: push up (landing/spawned inside)
247 playerTransform.Position.y += overlap.y;
248 if(m_PlayerController->GetVelocity().y < 0.0f){
249 m_PlayerController->GetVelocity().y = 0.0f;
250 }
251 grounded = true;
252 } else{
253 //Player center is below platform center: push down (hit from underneath)
254 playerTransform.Position.y -= overlap.y;
255 if(m_PlayerController->GetVelocity().y > 0.0f){
256 m_PlayerController->GetVelocity().y = 0.0f;
257 }
258 }
259 }
260 }
261 }
262 }
263
264 //Clamp to world floor
265 if(!grounded){
266 float worldFloor = m_World->GetMinY() + playerHalfSize.y;
267 if(playerTransform.Position.y <= worldFloor + GROUND_MARGIN){
268 playerTransform.Position.y = worldFloor;
269 if(m_PlayerController->GetVelocity().y <= 0.0f){
270 m_PlayerController->GetVelocity().y = 0.0f;
271 }
272 grounded = true;
273 }
274 }
275
276 m_PlayerController->SetGrounded(grounded);
277}

285void BaseLevelScene::CheckEnemyCollisions(){
286 auto& registry = m_ECSScene->GetRegistry();
287 auto& playerTransform = registry.get<Components::Transform>(m_PlayerEntity);
288 Vector3D playerHalf = playerTransform.Scale * 0.5f;
289 float playerBottom = playerTransform.Position.y - playerHalf.y;
290
291 auto enemyView = registry.view<Components::Transform, Components::Enemy,
292 Components::PhysicsBody>();
293
294 for(auto [entity, transform, enemy, physics] : enemyView.each()){
295 if(!enemy.IsActive) continue;
296
297 Vector3D enemyHalf = transform.Scale * 0.5f;
298 float dx = std::abs(playerTransform.Position.x - transform.Position.x);
299 float dy = std::abs(playerTransform.Position.y - transform.Position.y);
300 bool colliding = (dx < (playerHalf.x + enemyHalf.x)) &&
301 (dy < (playerHalf.y + enemyHalf.y));
302 if(!colliding) continue;
303
304 float enemyTop = transform.Position.y + enemyHalf.y;
305 bool isAbove = (playerBottom < enemyTop + 0.1f) &&
306 (playerBottom > enemyTop - 0.2f);
307 bool isFalling = (m_PlayerController->GetVelocity().y < 0.0f);
308
309 if(isFalling && isAbove){
310 //Stomp enemy
311 DestroyEnemy(entity);
312 m_PlayerController->GetVelocity().y = 5.0f;
313 Log::Info("Enemy stomped!");
314 } else{
315 //Player dies and all enemies respawn
316 m_PlayerController->Die("Killed by enemy");
317 RespawnEnemies();
318 break;
319 }
320 }
321}

213void Demo2DFixedScene::ResolveCollisions(){
214 auto& registry = m_ECSScene->GetRegistry();
215 auto& playerTransform = registry.get<Components::Transform>(m_PlayerEntity);
216 Vector3D playerHalfSize = playerTransform.Scale * 0.5f;
217
218 bool grounded = false;
219 const float GROUND_MARGIN = 0.1f;
220
221 float playerLeft = playerTransform.Position.x - playerHalfSize.x;
222 float playerRight = playerTransform.Position.x + playerHalfSize.x;
223 float playerBottom = playerTransform.Position.y - playerHalfSize.y;
224 float playerTop = playerTransform.Position.y + playerHalfSize.y;
225
226 for(entt::entity platform : m_PlatformEntities){
227 auto& platformTransform = registry.get<Components::Transform>(platform);
228 Vector3D platformHalfSize = platformTransform.Scale * 0.5f;
229
230 float platformLeft = platformTransform.Position.x - platformHalfSize.x;
231 float platformRight = platformTransform.Position.x + platformHalfSize.x;
232 float platformTop = platformTransform.Position.y + platformHalfSize.y;
233 float platformBottom = platformTransform.Position.y - platformHalfSize.y;
234
235 bool overlapX = (playerRight > platformLeft && playerLeft < platformRight);
236 bool overlapY = (playerTop > platformBottom && playerBottom < platformTop);
237
238 //Check if player is landing on top of platform
239 if(overlapX){
240 float distanceToTop = playerBottom - platformTop;
241 if(distanceToTop <= GROUND_MARGIN && distanceToTop >= -GROUND_MARGIN){
242 if(m_PlayerController->GetVelocity().y <= 0.0f){
243 grounded = true;
244 playerTransform.Position.y = platformTop + playerHalfSize.y;
245 m_PlayerController->GetVelocity().y = 0.0f;
246 break;
247 }
248 }
249 }
250
251 //General collision resolution
252 if(!grounded && overlapX && overlapY){
253 Vector3D delta = playerTransform.Position - platformTransform.Position;
254 Vector3D overlap = playerHalfSize + platformHalfSize - Vector3D(std::abs(delta.x), std::abs(delta.y), 0.0f);
255
256 if(overlap.x > 0.0f && overlap.y > 0.0f){
257 if(overlap.x < overlap.y){
258 //Horizontal collision
259 if(delta.x > 0.0f) playerTransform.Position.x += overlap.x;
260 else playerTransform.Position.x -= overlap.x;
261 } else{
262 //Vertical collision
263 if(delta.y < 0.0f){
264 playerTransform.Position.y -= overlap.y;
265 if(m_PlayerController->GetVelocity().y > 0.0f){
266 m_PlayerController->GetVelocity().y = 0.0f;
267 }
268 }
269 }
270 }
271 }
272 }
273
274 //Clamp to world floor
275 if(!grounded){
276 float worldFloor = m_World->GetMinY() + playerHalfSize.y;
277 if(playerTransform.Position.y <= worldFloor + GROUND_MARGIN){
278 playerTransform.Position.y = worldFloor;
279 if(m_PlayerController->GetVelocity().y <= 0.0f){
280 m_PlayerController->GetVelocity().y = 0.0f;
281 }
282 grounded = true;
283 }
284 }
285
286 m_PlayerController->SetGrounded(grounded);
287}

385void Demo2DFixedScene::CheckEnemyCollisions(){
386 auto& registry = m_ECSScene->GetRegistry();
387 auto& playerTransform = registry.get<Components::Transform>(m_PlayerEntity);
388 Vector3D playerSize = playerTransform.Scale;
389 Vector3D playerHalf = playerSize * 0.5f;
390 float playerBottom = playerTransform.Position.y - playerHalf.y;
391
392 auto enemyView = registry.view<Components::Transform, Components::Enemy, Components::PhysicsBody>();
393
394 for(auto [entity, transform, enemy, physics] : enemyView.each()){
395 if(!enemy.IsActive) continue;
396
397 Vector3D enemyHalf = transform.Scale * 0.5f;
398
399 //AABB collision check
400 float dx = std::abs(playerTransform.Position.x - transform.Position.x);
401 float dy = std::abs(playerTransform.Position.y - transform.Position.y);
402 bool colliding = (dx < (playerHalf.x + enemyHalf.x)) &&
403 (dy < (playerHalf.y + enemyHalf.y));
404
405 if(!colliding) continue;
406
407 //Determine collision direction
408 //Check if player is falling and the collision is from above the enemy - "stomping"
409 float enemyTop = transform.Position.y + enemyHalf.y;
410 bool isAbove = (playerBottom < enemyTop + 0.1f) && (playerBottom > enemyTop - 0.2f);
411 bool isFalling = (m_PlayerController->GetVelocity().y < 0.0f);
412
413 if(isFalling && isAbove){
414 //Stomp enemy and bounce
415 DestroyEnemy(entity);
416 m_PlayerController->GetVelocity().y = 5.0f;
417 Log::Info("Enemy stomped!");
418 } else{
419 //Player dies and respawns all enemies
420 Log::Info("Player collided with enemy!");
421 m_PlayerController->Die("Killed by enemy");
422 RespawnEnemies();
423 break;
424 }
425 }
426}

53void PlayerController::Update(){

59 //Get direction from keyboard

62 if(Input::IsKeyHeld(SDL_SCANCODE_RIGHT)) keyboardDirection.x += 1.0f;
63
64 //Normalize diagonal movement

74
75 float deadzone = 0.2f;
76 if(std::abs(gamepadDirection.x) < deadzone) gamepadDirection.x = 0.0f;
77 if(std::abs(gamepadDirection.y) < deadzone) gamepadDirection.y = 0.0f;
78
79 float magnitude = std::sqrt(gamepadDirection.x * gamepadDirection.x + gamepadDirection.y * gamepadDirection.y);
80 if(magnitude > 0.1f){

83 }
84 }
85

100 if(Input::IsKeyPressed(SDL_SCANCODE_SPACE)) wantToJump = true;

120 m_JumpTimer += dt;
121 float gravityReduction = 1.0f - (m_JumpTimer / 0.5f);
122 if(gravityReduction < 0.0f) gravityReduction = 0.0f;

130 else m_Velocity.y = 0.0f;
131 }

135
136 //Landing detection and sound
137 if(m_IsGrounded && !m_WasGrounded){
138 SoundManager::GetInstance().PlaySound("assets/sounds/landing.wav");
139 Log::Info("Landed (sound played)");
140 }

142
143 //Clamp player position to world boundaries

145
146 //Log player state for debugging

149
150 //Death check - fell off the world

152
153 if(m_IsGrounded) m_Velocity.y = 0.0f;

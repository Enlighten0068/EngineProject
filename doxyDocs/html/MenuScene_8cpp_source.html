209void MenuScene::RenderOption(const MenuOption& option, bool isSelected){
210 bool isHovered = IsMouseOverOption(option);
211 std::shared_ptr<Texture2D> texture = (isSelected || isHovered)
212 ? option.SelectedTexture
213 : option.NormalTexture;
214
215 if(!texture){
216 Log::Warning("RenderOption: texture is null for " + option.Label);
217 return;
218 }
219
220 //Build model matrix
221 Matrix4 model = Matrix4::Translation(option.Position);
222 float aspect = static_cast<float>(texture->GetWidth()) / static_cast<float>(texture->GetHeight());
223 float height = 1.5f;
224 float width = height * aspect;
225 model = model * Matrix4::Scale(Vector3D(width, height, 1.0f));
226

230
231 Renderer::DrawTexturedQuad(m_Shader, m_VertexArray, m_IndexBuffer,

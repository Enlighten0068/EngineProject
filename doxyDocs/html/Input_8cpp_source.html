47void Input::ProcessEvent(const SDL_Event& event){
48 switch(event.type){

50 s_MousePosition.x = static_cast<float>(event.motion.x);
51 s_MousePosition.y = static_cast<float>(event.motion.y);
52 s_MouseDelta.x += static_cast<float>(event.motion.xrel);
53 s_MouseDelta.y += static_cast<float>(event.motion.yrel);
54 break;
55 case SDL_EVENT_MOUSE_BUTTON_DOWN:
56 if(event.button.button >= 1 && event.button.button <= 5){
57 s_MouseButtonState[event.button.button - 1] = true;
58 }
59 break;
60 case SDL_EVENT_MOUSE_BUTTON_UP:
61 if(event.button.button >= 1 && event.button.button <= 5){
62 s_MouseButtonState[event.button.button - 1] = false;
63 }
64 break;
65 case SDL_EVENT_MOUSE_WHEEL:
66 s_ScrollDelta.x += static_cast<float>(event.wheel.x);
67 s_ScrollDelta.y += static_cast<float>(event.wheel.y);
68 break;
69 case SDL_EVENT_WINDOW_RESIZED:
70 s_WindowResized = true;
71 s_WindowWidth = event.window.data1;
72 s_WindowHeight = event.window.data2;

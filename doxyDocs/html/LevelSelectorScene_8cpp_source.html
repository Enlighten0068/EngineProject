173bool LevelSelectorScene::IsMouseOverOption(const Option& option) const{
174 Vector2D mousePos = Input::GetMousePosition();
175 int winWidth = Input::GetWindowWidth();
176 int winHeight = Input::GetWindowHeight();
177 float worldX = (mousePos.x / winWidth) * 20.0f - 10.0f;
178 float worldY = -(mousePos.y / winHeight) * 20.0f + 10.0f;
179
180 float halfWidth = (option.Width / option.Height) * 1.2f * 0.5f;
181 float halfHeight = 0.6f;
182
183 float left = option.Position.x - halfWidth;
184 float right = option.Position.x + halfWidth;
185 float bottom = option.Position.y - halfHeight;
186 float top = option.Position.y + halfHeight;
187
188 return (worldX >= left && worldX <= right && worldY >= bottom && worldY <= top);
189}

196void LevelSelectorScene::RenderOption(const Option& option, bool isSelected){

198 auto texture = (isSelected || isHovered) ? option.SelectedTexture : option.NormalTexture;
199 if(!texture) return;

201 Matrix4 model = Matrix4::Translation(option.Position);

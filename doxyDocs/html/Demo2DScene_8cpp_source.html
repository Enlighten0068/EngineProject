176void Demo2DScene::ResolveCollisions(){
177 auto& registry = m_ECSScene->GetRegistry();
178 auto& playerTransform = registry.get<Components::Transform>(m_PlayerEntity);
179 Vector3D playerHalfSize = playerTransform.Scale * 0.5f;
180
181 bool grounded = false;
182 const float GROUND_MARGIN = 0.1f;
183
184 float playerLeft = playerTransform.Position.x - playerHalfSize.x;
185 float playerRight = playerTransform.Position.x + playerHalfSize.x;
186 float playerBottom = playerTransform.Position.y - playerHalfSize.y;
187 float playerTop = playerTransform.Position.y + playerHalfSize.y;
188
189 for(entt::entity platform : m_PlatformEntities){
190 auto& platformTransform = registry.get<Components::Transform>(platform);
191 Vector3D platformHalfSize = platformTransform.Scale * 0.5f;
192
193 float platformLeft = platformTransform.Position.x - platformHalfSize.x;
194 float platformRight = platformTransform.Position.x + platformHalfSize.x;
195 float platformTop = platformTransform.Position.y + platformHalfSize.y;
196 float platformBottom = platformTransform.Position.y - platformHalfSize.y;
197
198 bool overlapX = (playerRight > platformLeft && playerLeft < platformRight);
199 bool overlapY = (playerTop > platformBottom && playerBottom < platformTop);
200
201 //Check if player is landing on top of platform
202 if(overlapX){
203 float distanceToTop = playerBottom - platformTop;
204 if(distanceToTop <= GROUND_MARGIN && distanceToTop >= -GROUND_MARGIN){
205 if(m_PlayerController->GetVelocity().y <= 0.0f){
206 grounded = true;
207 playerTransform.Position.y = platformTop + playerHalfSize.y;
208 m_PlayerController->GetVelocity().y = 0.0f;
209 break;
210 }
211 }
212 }
213
214 //General collision resolution
215 if(!grounded && overlapX && overlapY){
216 Vector3D delta = playerTransform.Position - platformTransform.Position;
217 Vector3D overlap = playerHalfSize + platformHalfSize - Vector3D(std::abs(delta.x), std::abs(delta.y), 0.0f);
218
219 if(overlap.x > 0.0f && overlap.y > 0.0f){
220 if(overlap.x < overlap.y){
221 //Horizontal collision
222 if(delta.x > 0.0f) playerTransform.Position.x += overlap.x;
223 else playerTransform.Position.x -= overlap.x;
224 } else{
225 //Vertical collision
226 if(delta.y < 0.0f){
227 playerTransform.Position.y -= overlap.y;
228 if(m_PlayerController->GetVelocity().y > 0.0f){
229 m_PlayerController->GetVelocity().y = 0.0f;
230 }
231 }
232 }
233 }
234 }
235 }
236
237 //Clamp to world floor
238 if(!grounded){
239 float worldFloor = m_World->GetMinY() + playerHalfSize.y;
240 if(playerTransform.Position.y <= worldFloor + GROUND_MARGIN){
241 playerTransform.Position.y = worldFloor;
242 if(m_PlayerController->GetVelocity().y <= 0.0f){
243 m_PlayerController->GetVelocity().y = 0.0f;
244 }
245 grounded = true;
246 }
247 }
248
249 m_PlayerController->SetGrounded(grounded);
250}
